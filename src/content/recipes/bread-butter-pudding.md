prepMins: 25
cookMins: 40
serves: 6
ingredients:
  - "8 slices white bread"
  - "50 g butter, softened"
  - "3 tbsp marmalade"
  - "50 g sultanas"
  - "3 eggs"
  - "300 ml milk"
  - "150 ml double cream"
  - "50 g caster sugar"
steps:
  - "Butter the bread, spread with marmalade and cut into triangles."
  - "Layer in a dish with the sultanas."
  - "Whisk the eggs, milk, cream and sugar, pour over and leave for 20 minutes."
  - "Bake at 180°C / 160°C fan for 35–40 minutes until set and golden."
