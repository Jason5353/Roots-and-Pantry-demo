prepMins: 20
cookMins: 150
serves: 10
ingredients:
  - "2 kg gammon joint"
  - "1 onion, halved"
  - "4 tbsp runny honey"
  - "2 tbsp marmalade"
  - "1 tbsp wholegrain mustard"
  - "20 cloves"
steps:
  - "Simmer the gammon with the onion for 20 minutes per 500 g."
  - "Heat the oven to 200°C / 180°C fan."
  - "Score the fat into diamonds and stud with cloves."
  - "Mix the honey, marmalade and mustard and spread over the fat."
  - "Roast for 30 minutes, basting halfway."
